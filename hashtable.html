Hashtable

A key gets hashed to a number, then dropped into bucket = hash % size. Two keys landing on the same bucket is a collision — this table resolves it by chaining, and resizes once it gets crowded.

How it works

Hashing turns an arbitrary key (here, a string) into a number, using a formula that mixes every character together — this page multiplies a running total by 31 and adds each character's code, a common simple approach. That raw number is then reduced with % size (size = number of buckets) so it always lands on a valid bucket index, no matter how large the original hash was.

Collisions happen when two different keys hash to the same bucket — with enough keys and a fixed number of buckets, this is mathematically unavoidable (the pigeonhole principle). This table handles it with chaining: each bucket holds a small list, and a colliding key just gets appended to that bucket's list instead of overwriting anything. Looking up a key means hashing it to find the right bucket, then scanning that bucket's short list.

Load factor (entries ÷ buckets) measures how crowded the table is getting. As it climbs, chains get longer and lookups get slower — approaching O(n) in the worst case instead of the O(1) average a hash table is supposed to give you. Once load factor crosses a threshold (0.75 here), the table resizes: it allocates more buckets and rehashes every existing key into the new, larger table, since hash % size gives a different answer once size changes.

function hash(key, size):
    h <- 0
    for each character c in key:
        h <- (h * 31 + charCode(c)) mod size   // mix each character into a running hash
    return h                                    // final bucket index, always in [0, size)

function insert(table, key, value):
    index <- hash(key, table.size)              // find which bucket this key belongs to
    table.buckets[index].append((key, value))   // chaining: append, don't overwrite

function get(table, key):
    index <- hash(key, table.size)              // find the same bucket used at insert time
    for (k, v) in table.buckets[index]:          // scan the (short) chain in that bucket
        if k == key: return v                     // match found
    return not found                            // exhausted the chain without a match
class HashTable:
    def __init__(self, size=8):
        self.size = size
        self.buckets = [[] for _ in range(size)]   # one empty chain per bucket

    def _hash(self, key):
        h = 0
        for ch in key:
            h = (h * 31 + ord(ch)) % self.size   # mix each character into a running hash
        return h                                  # bucket index, always in [0, size)

    def insert(self, key, value):
        index = self._hash(key)               # find which bucket this key belongs to
        for i, (k, v) in enumerate(self.buckets[index]):  # check if the key already exists
            if k == key:
                self.buckets[index][i] = (key, value)  # update existing entry in place
                return
        self.buckets[index].append((key, value))  # new key: append to the chain

    def get(self, key):
        index = self._hash(key)             # find the same bucket used at insert time
        for k, v in self.buckets[index]:      # scan the (short) chain in that bucket
            if k == key:
                return v                        # match found
        raise KeyError(key)                  # exhausted the chain without a match
class HashTable {
  constructor(size = 8) {
    this.size = size;
    this.buckets = Array.from({ length: size }, () => []); // one empty chain per bucket
  }

  hash(key) {
    let h = 0;
    for (const ch of key) h = (h * 31 + ch.charCodeAt(0)) % this.size; // mix each char in
    return h;   // bucket index, always in [0, size)
  }

  insert(key, value) {
    const index = this.hash(key);              // find which bucket this key belongs to
    const bucket = this.buckets[index];
    const existing = bucket.find(([k]) => k === key);  // check if the key already exists
    if (existing) existing[1] = value;           // update existing entry in place
    else bucket.push([key, value]);              // new key: append to the chain
  }

  get(key) {
    const index = this.hash(key);   // find the same bucket used at insert time
    const found = this.buckets[index].find(([k]) => k === key); // scan the chain
    return found ? found[1] : undefined;   // undefined if the chain had no match
  }
}
class HashTable {
    private LinkedList<Entry>[] buckets;
    private int size;

    HashTable(int size) {
        this.size = size;
        buckets = new LinkedList[size];
        for (int i = 0; i < size; i++) buckets[i] = new LinkedList<>(); // empty chains
    }

    private int hash(String key) {
        int h = 0;
        for (char c : key.toCharArray()) h = (h * 31 + c) % size; // mix each char in
        return h;   // bucket index, always in [0, size)
    }

    void insert(String key, String value) {
        int index = hash(key);                 // find which bucket this key belongs to
        for (Entry e : buckets[index]) {         // check if the key already exists
            if (e.key.equals(key)) { e.value = value; return; } // update in place
        }
        buckets[index].add(new Entry(key, value)); // new key: append to the chain
    }

    String get(String key) {
        for (Entry e : buckets[hash(key)]) {     // scan the chain in the right bucket
            if (e.key.equals(key)) return e.value; // match found
        }
        return null;   // exhausted the chain without a match
    }
}
class HashTable {
    vector<list<pair<string,string>>> buckets;   // one chain per bucket
    int size;

    int hashKey(const string& key) {
        int h = 0;
        for (char c : key) h = (h * 31 + c) % size;  // mix each char into a running hash
        return h;   // bucket index, always in [0, size)
    }

public:
    HashTable(int size) : size(size), buckets(size) {}   // pre-allocate empty chains

    void insert(const string& key, const string& value) {
        int index = hashKey(key);              // find which bucket this key belongs to
        for (auto& [k, v] : buckets[index]) {    // check if the key already exists
            if (k == key) { v = value; return; }   // update in place
        }
        buckets[index].push_back({key, value}); // new key: append to the chain
    }

    string get(const string& key) {
        for (auto& [k, v] : buckets[hashKey(key)]) {  // scan the chain in the right bucket
            if (k == key) return v;                     // match found
        }
        return "";   // exhausted the chain without a match
    }
};
