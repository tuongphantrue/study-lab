LRU Cache

A hash map for O(1) lookup, plus an ordering for O(1) eviction. Access a key and watch it jump to the front — fill the cache past capacity and watch the oldest key get evicted.

How it works

An LRU (least-recently-used) cache needs two operations to be fast: look up a key, and know which item was touched longest ago. Neither structure alone gives you both cheaply — a hash map gives O(1) lookup but no ordering; a plain list gives ordering but O(n) lookup — so the cache combines them.

A doubly linked list keeps every key in order from most- to least-recently used. A hash map stores key → node, so any key can jump straight to its spot in the list in O(1), instead of scanning for it.

On every get or put, the touched node is unlinked and moved to the front (most-recent end) — O(1) because doubly-linked nodes know their own neighbors. When the cache is full and a new key arrives, the node at the back (least-recently used) is evicted.

from collections import OrderedDict   # remembers insertion/access order

class LRUCache:
    def __init__(self, capacity):
        self.capacity = capacity
        self.cache = OrderedDict()   # front = least recent, back = most recent

    def get(self, key):
        if key not in self.cache:
            return -1                        # key was never cached, or already evicted
        self.cache.move_to_end(key)           # mark as most recently used
        return self.cache[key]

    def put(self, key, value):
        if key in self.cache:
            self.cache.move_to_end(key)         # already cached: refresh its position
        self.cache[key] = value                # insert or overwrite
        if len(self.cache) > self.capacity:
            self.cache.popitem(last=False)      # over capacity: evict the least recently used
class LRUCache {
  constructor(capacity) {
    this.capacity = capacity;
    this.cache = new Map();   // Maps preserve insertion order in JS, which we exploit here
  }

  get(key) {
    if (!this.cache.has(key)) return -1;   // key was never cached, or already evicted
    const value = this.cache.get(key);
    this.cache.delete(key);                 // remove and re-add to move it to the "recent" end
    this.cache.set(key, value);              // now it's the most recently used entry
    return value;
  }

  put(key, value) {
    if (this.cache.has(key)) this.cache.delete(key);   // remove old position first, if any
    this.cache.set(key, value);                          // insert as most recently used
    if (this.cache.size > this.capacity) {
      const oldest = this.cache.keys().next().value;      // Map iterates oldest-first
      this.cache.delete(oldest);                            // evict least recently used
    }
  }
}
class LRUCache extends LinkedHashMap<Integer, Integer> {
    private int capacity;

    LRUCache(int capacity) {
        super(capacity, 0.75f, true);   // accessOrder=true reorders entries on every get()
        this.capacity = capacity;
    }

    @Override
    protected boolean removeEldestEntry(Map.Entry<Integer, Integer> eldest) {
        return size() > capacity;   // returning true here auto-evicts the least recently used
    }
}

class LRUCache {
    int capacity;
    list<pair<int,int>> order;   // doubly linked list; front = most recently used
    unordered_map<int, list<pair<int,int>>::iterator> map;   // key -> position in the list

public:
    LRUCache(int capacity) : capacity(capacity) {}

    int get(int key) {
        if (!map.count(key)) return -1;               // key was never cached, or evicted
        order.splice(order.begin(), order, map[key]);   // move this node to the front, O(1)
        return map[key]->second;
    }

    void put(int key, int value) {
        if (map.count(key)) order.erase(map[key]);      // remove old entry first, if present
        order.push_front({key, value});                 // insert as most recently used
        map[key] = order.begin();                         // remember its new position
        if ((int)map.size() > capacity) {
            auto last = order.back();                       // the least recently used entry
            map.erase(last.first);
            order.pop_back();                                 // evict it
        }
    }
};
