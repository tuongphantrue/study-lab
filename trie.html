Trie (Prefix Tree)

Insert words and watch shared prefixes branch off the same path. Type a prefix below to see live autocomplete pulled straight from the tree.

How it works

A trie (prefix tree) stores words letter by letter, with shared prefixes sharing the same path from the root. Inserting "cat" then "car" creates one shared branch for c → a, then splits into t and r.

Each node marks whether a word ends there — that's how the trie distinguishes "car" (a complete word) from "ca" (just a prefix on the way to something else) even though both are valid paths through the tree.

Autocomplete works by walking down to the node matching the typed prefix, then collecting every complete word reachable below that point — which is why it's fast: you only ever explore the subtree under your prefix, never the whole dictionary.

class TrieNode:
    def __init__(self):
        self.children = {}    # letter -> child TrieNode
        self.is_end = False    # True if a word ends exactly at this node

class Trie:
    def __init__(self):
        self.root = TrieNode()   # the empty-prefix starting point

    def insert(self, word):
        node = self.root
        for ch in word:
            node = node.children.setdefault(ch, TrieNode())  # create branch if missing
        node.is_end = True   # mark the final node as a complete word

    def starts_with(self, prefix):
        node = self.root
        for ch in prefix:
            if ch not in node.children:
                return []       # prefix isn't in the trie at all
            node = node.children[ch]   # descend one letter at a time
        return self._collect(node, prefix)   # gather every word below this point

    def _collect(self, node, prefix):
        words = [prefix] if node.is_end else []   # this node itself might be a word
        for ch, child in node.children.items():     # recurse into every branch
            words += self._collect(child, prefix + ch)
        return words
class TrieNode {
  constructor() {
    this.children = {};   // letter -> child TrieNode
    this.isEnd = false;    // true if a word ends exactly at this node
  }
}

class Trie {
  constructor() { this.root = new TrieNode(); }   // the empty-prefix starting point

  insert(word) {
    let node = this.root;
    for (const ch of word) {
      if (!node.children[ch]) node.children[ch] = new TrieNode(); // create branch if missing
      node = node.children[ch];   // descend into that branch
    }
    node.isEnd = true;   // mark the final node as a complete word
  }

  startsWith(prefix) {
    let node = this.root;
    for (const ch of prefix) {
      if (!node.children[ch]) return [];   // prefix isn't in the trie at all
      node = node.children[ch];             // descend one letter at a time
    }
    return this.collect(node, prefix);   // gather every word below this point
  }

  collect(node, prefix) {
    let words = node.isEnd ? [prefix] : [];   // this node itself might be a word
    for (const ch in node.children) {           // recurse into every branch
      words = words.concat(this.collect(node.children[ch], prefix + ch));
    }
    return words;
  }
}
class TrieNode {
    Map<Character, TrieNode> children = new HashMap<>();   // letter -> child node
    boolean isEnd = false;    // true if a word ends exactly at this node
}

class Trie {
    TrieNode root = new TrieNode();   // the empty-prefix starting point

    void insert(String word) {
        TrieNode node = root;
        for (char ch : word.toCharArray()) {
            node = node.children.computeIfAbsent(ch, c -> new TrieNode()); // create if missing
        }
        node.isEnd = true;   // mark the final node as a complete word
    }

    List<String> startsWith(String prefix) {
        TrieNode node = root;
        for (char ch : prefix.toCharArray()) {
            if (!node.children.containsKey(ch)) return new ArrayList<>(); // not in the trie
            node = node.children.get(ch);   // descend one letter at a time
        }
        List<String> results = new ArrayList<>();
        collect(node, prefix, results);   // gather every word below this point
        return results;
    }

    void collect(TrieNode node, String prefix, List<String> results) {
        if (node.isEnd) results.add(prefix);   // this node itself might be a word
        for (var entry : node.children.entrySet()) {   // recurse into every branch
            collect(entry.getValue(), prefix + entry.getKey(), results);
        }
    }
}
struct TrieNode {
    unordered_map<char, TrieNode*> children;   // letter -> child node
    bool isEnd = false;    // true if a word ends exactly at this node
};

class Trie {
    TrieNode* root = new TrieNode();   // the empty-prefix starting point

    void collect(TrieNode* node, string prefix, vector<string>& out) {
        if (node->isEnd) out.push_back(prefix);   // this node itself might be a word
        for (auto& [ch, child] : node->children) {  // recurse into every branch
            collect(child, prefix + ch, out);
        }
    }

public:
    void insert(const string& word) {
        TrieNode* node = root;
        for (char ch : word) {
            if (!node->children.count(ch)) node->children[ch] = new TrieNode(); // create if missing
            node = node->children[ch];   // descend into that branch
        }
        node->isEnd = true;   // mark the final node as a complete word
    }

    vector<string> startsWith(const string& prefix) {
        TrieNode* node = root;
        for (char ch : prefix) {
            if (!node->children.count(ch)) return {};   // prefix isn't in the trie at all
            node = node->children[ch];                    // descend one letter at a time
        }
        vector<string> results;
        collect(node, prefix, results);   // gather every word below this point
        return results;
    }
};
